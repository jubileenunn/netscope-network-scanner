import ipaddress
import platform
import sqlite3
import subprocess
import sys
import time
from datetime import datetime
from typing import List, Optional, Tuple


DB_NAME = "netscope_inventory.db"


# ============================================================
# Database Layer
# ============================================================

def init_db(db_name: str = DB_NAME) -> None:
    """Create the assets table if it doesn't exist."""
    conn = sqlite3.connect(db_name)
    cur = conn.cursor()

    cur.execute(""" 
        CREATE TABLE IF NOT EXISTS assets ( 
            id INTEGER PRIMARY KEY AUTOINCREMENT, 
            ip_address TEXT NOT NULL, 
            hostname TEXT, 
            mac_address TEXT, 
            vendor TEXT, 
            last_seen TEXT NOT NULL 
        ) 
    """)

    conn.commit()
    conn.close()


def upsert_asset(
    ip_address: str,
    hostname: Optional[str],
    mac_address: Optional[str],
    vendor: Optional[str],
    db_name: str = DB_NAME,
) -> None:
    """Insert or update an asset entry."""
    conn = sqlite3.connect(db_name)
    cur = conn.cursor()

    now = datetime.utcnow().isoformat()

    cur.execute("SELECT id FROM assets WHERE ip_address = ?", (ip_address,))
    row = cur.fetchone()

    if row:
        # Update existing asset
        cur.execute(""" 
            UPDATE assets 
            SET hostname = COALESCE(?, hostname), 
                mac_address = COALESCE(?, mac_address), 
                vendor = COALESCE(?, vendor), 
                last_seen = ? 
            WHERE id = ? 
        """, (hostname, mac_address, vendor, now, row[0]))
    else:
        # Insert new asset
        cur.execute(""" 
            INSERT INTO assets (ip_address, hostname, mac_address, vendor, last_seen) 
            VALUES (?, ?, ?, ?, ?) 
        """, (ip_address, hostname, mac_address, vendor, now))

    conn.commit()
    conn.close()


def list_assets(db_name: str = DB_NAME) -> List[Tuple]:
    """Return all assets from the database."""
    conn = sqlite3.connect(db_name)
    cur = conn.cursor()

    cur.execute(""" 
        SELECT ip_address, hostname, mac_address, vendor, last_seen 
        FROM assets 
        ORDER BY ip_address 
    """)

    rows = cur.fetchall()
    conn.close()
    return rows


# ============================================================
# Network Utilities
# ============================================================

def ping_host(ip: str, timeout: int = 1000) -> bool:
    """Return True if host responds to ping."""
    system = platform.system().lower()

    if system == "windows":
        cmd = ["ping", "-n", "1", "-w", str(timeout), ip]
    else:
        cmd = ["ping", "-c", "1", "-W", str(int(timeout / 1000)), ip]

    try:
        result = subprocess.run(
            cmd,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
        )
        return result.returncode == 0
    except Exception:
        return False


def get_arp_table() -> str:
    """Return raw ARP table output."""
    system = platform.system().lower()
    cmd = ["arp", "-a"] if system == "windows" else ["arp", "-n"]

    try:
        result = subprocess.run(
            cmd,
            stdout=subprocess.PIPE,
            stderr=subprocess.DEVNULL,
            text=True,
        )
        return result.stdout
    except Exception:
        return ""


def parse_arp_table(arp_output: str) -> dict:
    """Parse ARP output into a dict of {ip: mac}."""
    mapping = {}

    for line in arp_output.splitlines():
        parts = line.split()
        if len(parts) < 3:
            continue

        ip_candidate = parts[0]
        mac_candidate = parts[1]

        # Validate IP
        try:
            ipaddress.ip_address(ip_candidate)
        except ValueError:
            continue

        # Validate MAC (very loose)
        if ":" in mac_candidate or "-" in mac_candidate:
            mapping[ip_candidate] = mac_candidate

    return mapping


def resolve_hostname(ip: str) -> Optional[str]:
    """Attempt to resolve hostname via nslookup."""
    try:
        result = subprocess.run(
            ["nslookup", ip],
            stdout=subprocess.PIPE,
            stderr=subprocess.DEVNULL,
            text=True,
        )

        for line in result.stdout.splitlines():
            if "name =" in line.lower():
                return line.split("=", 1)[1].strip().strip(".")
    except Exception:
        pass

    return None


def lookup_vendor_from_mac(mac: str) -> Optional[str]:
    """Placeholder for MAC vendor lookup."""
    return None


# ============================================================
# Network Scanner
# ============================================================

def scan_network(cidr: str) -> None:
    """Ping sweep the network and record live hosts."""
    print(f"[+] Scanning network: {cidr}")

    network = ipaddress.ip_network(cidr, strict=False)
    arp_map = parse_arp_table(get_arp_table())

    for ip in network.hosts():
        ip_str = str(ip)

        if not ping_host(ip_str):
            continue

        print(f"  [*] Host up: {ip_str}")

        hostname = resolve_hostname(ip_str)
        mac = arp_map.get(ip_str)
        vendor = lookup_vendor_from_mac(mac) if mac else None

        upsert_asset(ip_str, hostname, mac, vendor)

        time.sleep(0.05)

    print("[+] Scan complete.")


# ============================================================
# CLI
# ============================================================

def print_assets() -> None:
    """Pretty-print all assets."""
    rows = list_assets()

    if not rows:
        print("No assets found in inventory yet.")
        return

    print(f"{'IP Address':<16} {'Hostname':<30} {'MAC Address':<20} {'Vendor':<20} {'Last Seen'}")
    print("-" * 110)

    for ip, hostname, mac, vendor, last_seen in rows:
        print(f"{ip:<16} {str(hostname or '-'): <30} {str(mac or '-'): <20} {str(vendor or '-'): <20} {last_seen}")


def main():
    if len(sys.argv) < 2:
        print("Usage:")
        print("  python netscope_inventory.py init-db")
        print("  python netscope_inventory.py scan <CIDR>")
        print("  python netscope_inventory.py list")
        sys.exit(1)

    command = sys.argv[1].lower()

    if command == "init-db":
        init_db()
        print(f"Initialized database: {DB_NAME}")

    elif command == "scan":
        if len(sys.argv) < 3:
            print("Usage: python netscope_inventory.py scan <CIDR>")
            sys.exit(1)

        init_db()
        scan_network(sys.argv[2])

    elif command == "list":
        init_db()
        print_assets()

    else:
        print(f"Unknown command: {command}")
        sys.exit(1)


if __name__ == "__main__":
    main()
